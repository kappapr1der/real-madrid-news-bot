# Импорты вверху digest.py:
from utils.source_map import map_source
from utils.time_labels import digest_label
from utils.title_extractor import extract_title

def _safe_title(item):
    title = (item.title or "").strip()
    if not title:
        title = extract_title(item.url) or ""
    return title.strip()

def _safe_source(item):
    return item.source or map_source(item.url)

def format_digest_entry(idx, item, translated_title_ru: str | None = None):
    title_ru = (translated_title_ru or _safe_title(item) or "").strip()
    if not title_ru:
        title_ru = "Без заголовка"
    source = _safe_source(item)
    title_ru = title_ru.replace('"Реал"', "«Реал»")
    return (
        f"{idx}\uFE0F\u20E3 {title_ru}\n"
        f"🔗 {item.url}\n"
        f"Источник: {source}"
    )
